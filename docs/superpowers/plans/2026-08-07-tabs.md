# Tab Support Implementation Plan

**Goal:** Add multiple tabs per window to chromeless, with the tab bar hidden whenever a
window holds a single tab, so the app looks untouched until tabs are actually used.

**Architecture:** Extract per-page state out of `BrowserWindowController` into a `Tab`
class that owns one `BrowserWebView`. The controller keeps an array of tabs plus an
active index, and exposes `webView` as a computed property returning the active tab's
web view — so every existing menu action, HUD path, snapshot, and download keeps
working unchanged. A hand-drawn `TabBarView` sits at the top edge, and the profile
badge becomes a clickable chip that docks into that bar.

**Tech Stack:** Swift 5, AppKit, WebKit. Single file `main.swift`, built by `./build.sh`
with `swiftc`. No test target — verification is build + manual drive.

**Spec:** `docs/superpowers/specs/2026-08-07-tabs-design.md`

---

## File structure

Everything lives in `main.swift` (1416 lines), matching the project's single-file
premise. New sections are inserted in the existing `// MARK:` layout:

| Section | Location | Responsibility |
| --- | --- | --- |
| `makeWebConfiguration(for:)` | after `// MARK: - Views` | Free function building a `WKWebViewConfiguration` for a profile — callable before `super.init` |
| `Tab` | after the above | One web view plus its page state |
| `ProfileChipView` | replaces `PassthroughVisualEffectView` | Clickable profile chip |
| `TabItemView` | after `ProfileChipView` | One tab: title, hover, close button |
| `TabBarView` | after `TabItemView` | Lays out tab items, `+` button, docked chip |
| `BrowserWindowController` | existing, modified | Owns `tabs`, `activeIndex`, tab commands |

private func observe(_ tab: Tab) {
    tab.observations = [
        tab.webView.observe(\.estimatedProgress, options: [.new]) { [weak self, weak tab] wv, _ in
            guard let self, let tab else { return }
            tab.lastProgress = CGFloat(wv.estimatedProgress)
            if tab === self.activeTab { self.progressChanged(wv.estimatedProgress) }
        },
        tab.webView.observe(\.title) { [weak self, weak tab] _, _ in
            guard let self, let tab else { return }
            tab.title = tab.displayTitle
            self.tabBar.update(titleAt: self.tabs.firstIndex { $0 === tab }, to: tab.title)
            if tab === self.activeTab { self.syncWindowTitle() }
        },
        tab.webView.observe(\.url) { [profile] wv, _ in
            if let u = wv.url, u.scheme == "https" || u.scheme == "http" {
                profileStore.recordVisit(u, for: profile)
            }
        },
    ]
}

private func syncWindowTitle() {
    let t = activeTab.webView.title ?? ""
    window?.title = "\(t.isEmpty ? "Chromeless" : t) - \(profile.name)"
}

final class TabBarView: NSVisualEffectView {
    static let height: CGFloat = 34
    static let trafficLightInset: CGFloat = 78

    var onSelect: ((Int) -> Void)?
    var onClose: ((Int) -> Void)?
    var onNewTab: (() -> Void)?

    private var items: [TabItemView] = []
    private let addButton = NSButton()
    private(set) var chipWidth: CGFloat = 0

    override init(frame frameRect: NSRect) {
        super.init(frame: frameRect)
        material = .hudWindow
        blendingMode = .withinWindow
        state = .active
        wantsLayer = true

        addButton.isBordered = false
        addButton.bezelStyle = .inline
        addButton.title = "+"
        addButton.font = .systemFont(ofSize: 15, weight: .medium)
        addButton.contentTintColor = .secondaryLabelColor
        addButton.target = self
        addButton.action = #selector(addClicked)
        addSubview(addButton)
    }

    required init?(coder: NSCoder) { fatalError("not used") }

    @objc private func addClicked() { onNewTab?() }

    func rebuild(titles: [String], activeIndex: Int) {
        for item in items { item.removeFromSuperview() }
        items = titles.enumerated().map { index, title in
            let item = TabItemView(frame: .zero)
            item.title = title
            item.isActive = index == activeIndex
            item.onSelect = { [weak self] in self?.onSelect?(index) }
            item.onClose = { [weak self] in self?.onClose?(index) }
            addSubview(item)
            return item
        }
        needsLayout = true
    }

    func update(titleAt index: Int?, to title: String) {
        guard let index, items.indices.contains(index) else { return }
        items[index].title = title
    }

    func setChipWidth(_ width: CGFloat) {
        chipWidth = width
        needsLayout = true
    }

    override func layout() {
        super.layout()
        let b = bounds
        let left = Self.trafficLightInset
        let rightReserve = chipWidth + 18
        let addW: CGFloat = 26
        let available = max(0, b.width - left - rightReserve - addW - 8)
        let count = max(1, items.count)
        let tabW = min(190, max(90, available / CGFloat(count)))

        var x = left
        for item in items {
            item.frame = NSRect(x: x, y: 3, width: tabW - 3, height: b.height - 6)
            x += tabW
        }
        addButton.frame = NSRect(x: min(x + 2, b.width - rightReserve - addW),
                                 y: (b.height - 22) / 2, width: addW, height: 22)
    }
}
```

private let tabBar = TabBarView()
private var keyMonitor: Any?

private var tabBarVisible: Bool { tabs.count > 1 }
private var tabBarHeight: CGFloat { tabBarVisible ? TabBarView.height : 0 }

private func makeTab(configuration: WKWebViewConfiguration? = nil) -> Tab {
    let conf = configuration ?? makeWebConfiguration(for: profile)
    let wv = BrowserWebView(frame: .zero, configuration: conf)
    wv.onEscape = { [weak self] in self?.escapeToStart() ?? false }
    wv.navigationDelegate = self
    wv.uiDelegate = self
    wv.allowsBackForwardNavigationGestures = true
    wv.allowsMagnification = true
    wv.underPageBackgroundColor = NSColor(calibratedWhite: 0.04, alpha: 1)
    if #available(macOS 13.3, *) { wv.isInspectable = true }
    let tab = Tab(webView: wv)
    observe(tab)
    return tab
}

@discardableResult
func addTab(url: URL?, configuration: WKWebViewConfiguration? = nil, activate: Bool = true) -> Tab {
    let tab = makeTab(configuration: configuration)
    tabs.append(tab)
    if activate {
        activeIndex = tabs.count - 1
    }
    refreshTabs()
    if configuration == nil {
        if let url { load(url, in: tab) } else { loadStartPage(in: tab) }
    }
    return tab
}

func selectTab(at index: Int) {
    guard tabs.indices.contains(index), index != activeIndex else { return }
    activeIndex = index
    refreshTabs()
}

func closeTab(at index: Int) {
    guard tabs.indices.contains(index) else { return }
    if tabs.count == 1 { window?.performClose(nil); return }
    let tab = tabs.remove(at: index)
    tab.teardown()
    if activeIndex >= tabs.count { activeIndex = tabs.count - 1 }
    else if index < activeIndex { activeIndex -= 1 }
    refreshTabs()
}

private func refreshTabs() {
    guard let container = window?.contentView else { return }
    for tab in tabs where tab.webView.superview != nil && tab !== activeTab {
        tab.webView.removeFromSuperview()
    }
    if activeTab.webView.superview == nil {
        container.addSubview(activeTab.webView, positioned: .below, relativeTo: tabBar)
    }
    tabBar.rebuild(titles: tabs.map(\.displayTitle), activeIndex: activeIndex)
    syncWindowTitle()
    lastProgressSync()
    layoutOverlays()
}

private func lastProgressSync() {
    let p = activeTab.lastProgress
    progressBar.alphaValue = (p > 0 && p < 1) ? 1 : 0
}
```

func load(_ url: URL, in tab: Tab) {
    tab.onStartPage = false
    if url.isFileURL {
        tab.webView.loadFileURL(url, allowingReadAccessTo: url.deletingLastPathComponent())
    } else {
        tab.webView.load(URLRequest(url: url))
    }

func loadStartPage() {

    }

func windowWillClose(_ notification: Notification) {
    if let monitor = mouseMonitor { NSEvent.removeMonitor(monitor) }
    if let monitor = keyMonitor { NSEvent.removeMonitor(monitor) }
    mouseMonitor = nil
    keyMonitor = nil
    for tab in tabs { tab.teardown() }
    tabs.removeAll()
    onClose?()
}

private func handleLoadError(_ error: Error, in webView: WKWebView) {
    let e = error as NSError
    if e.code == NSURLErrorCancelled || e.code == 102 { return }
    if launchOptions.snap != nil {
